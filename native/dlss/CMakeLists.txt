cmake_minimum_required(VERSION 3.24)
project(wynn_dlss LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The initial DLSS bridge targets Windows x64 with MSVC")
endif()
set(DLSS_SDK_ROOT "" CACHE PATH "NVIDIA DLSS SDK checkout (see README)")
set(VULKAN_HEADERS_ROOT "" CACHE PATH "Khronos Vulkan-Headers checkout")
if(NOT EXISTS "${DLSS_SDK_ROOT}/include/nvsdk_ngx_vk.h")
    message(FATAL_ERROR "Set DLSS_SDK_ROOT to the NVIDIA SDK checkout")
endif()
if(NOT EXISTS "${VULKAN_HEADERS_ROOT}/include/vulkan/vulkan.h")
    message(FATAL_ERROR "Set VULKAN_HEADERS_ROOT to the Vulkan-Headers checkout")
endif()
add_library(wynn_dlss SHARED src/wynn_dlss.cpp)
target_compile_definitions(wynn_dlss PRIVATE WV_DLSS_BUILD)
target_compile_features(wynn_dlss PRIVATE cxx_std_17)
target_include_directories(wynn_dlss PRIVATE include)
target_include_directories(wynn_dlss SYSTEM PRIVATE
    "${DLSS_SDK_ROOT}/include" "${VULKAN_HEADERS_ROOT}/include")
set_property(TARGET wynn_dlss PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_options(wynn_dlss PRIVATE /W4 /WX /EHsc)
target_link_libraries(wynn_dlss PRIVATE
    "${DLSS_SDK_ROOT}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib"
    shlwapi version)
# NVIDIA runtime binaries are intentionally not copied or packed into the mod.
add_executable(dlss_smoke tests/smoke.cpp)
target_compile_features(dlss_smoke PRIVATE cxx_std_17)
target_include_directories(dlss_smoke PRIVATE include "${VULKAN_HEADERS_ROOT}/include")
set_property(TARGET dlss_smoke PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_options(dlss_smoke PRIVATE /W4 /WX /EHsc)
target_link_libraries(dlss_smoke PRIVATE wynn_dlss)
